## 候选 Tick、索敌交接与宿主结果

~~~csharp
using System;
using System.Collections.Generic;
using ProjectB.BattleSpatial;

namespace ProjectB.BattleIntegration
{
    public enum TickDecision : byte { Continue, StopAfterCommit }

    public readonly struct BattleTick
    {
        public readonly long BattleId;
        public readonly long Number;
        public readonly int TickRate;

        public BattleTick(long battleId, long number, int tickRate)
        {
            BattleId = battleId;
            Number = number;
            TickRate = tickRate;
        }
    }

    public readonly ref struct TargetPhaseInput
    {
        public readonly BatchStamp Stamp;
        public readonly ReadOnlySpan<SpatialTarget> Targets;
        public readonly ReadOnlySpan<TargetQuery> Queries;
        public readonly ReadOnlySpan<EntityRef> Excluded;

        public TargetPhaseInput(BatchStamp stamp, ReadOnlySpan<SpatialTarget> targets,
            ReadOnlySpan<TargetQuery> queries, ReadOnlySpan<EntityRef> excluded)
        {
            Stamp = stamp;
            Targets = targets;
            Queries = queries;
            Excluded = excluded;
        }
    }

    public struct BattleTickResult
    {
        public long BattleId;
        public long Tick;
        public long SnapshotKey;
        public TickDecision Decision;
    }

    public interface IBattleStages : IDisposable
    {
        /// <summary>撤销已到期对象的行动、索敌与受击资格。</summary>
        void RevokeExpired(in BattleTick tick);

        /// <summary>封存并按稳定顺序提交本步核准指令。</summary>
        void CommitCommands(in BattleTick tick);

        /// <summary>固定保护、通行、有效属性与移动前位置。</summary>
        void FreezePreMove(in BattleTick tick);

        /// <summary>取本阶段冻结的目标与已复核旧锁后生成的查询。</summary>
        TargetPhaseInput CaptureTargeting(in BattleTick tick, TargetPhase phase);

        /// <summary>复核寻敌者、结果目标身份和此阶段的资格。</summary>
        bool CanApplyTarget(in BatchStamp stamp, in TargetResult result);

        /// <summary>由单位或建筑所有者按生成 API 提交本阶段目标。</summary>
        void ApplyTarget(in BatchStamp stamp, in TargetResult result);

        /// <summary>从相同起点采集输入、执行并完成 RVO2，然后统一提交位置。</summary>
        void StepRvo2AndCommitPositions(in BattleTick tick);

        /// <summary>固定移动后位置、受击形状和当步资格。</summary>
        void FreezePostMove(in BattleTick tick);

        /// <summary>按新位置复核发射，按稳定顺序固定请求与命中快照。</summary>
        void CollectAttacks(in BattleTick tick);

        /// <summary>统一结算，按所有权写入生命、破损及下一步保护。</summary>
        void ResolveAndCommit(in BattleTick tick);

        /// <summary>清理并产出由宿主持续持有的值快照引用。</summary>
        BattleTickResult Finish(in BattleTick tick);
    }

    /// <summary>由帧通道或宿主持有，生命周期覆盖 World 销毁后仍待消费的结果。</summary>
    public sealed class BattleOutbox
    {
        private readonly long battleId;
        private readonly int capacity;
        private readonly Queue<BattleTickResult> queue;

        public BattleOutbox(long battleId, int capacity)
        {
            if (battleId <= 0 || capacity <= 0)
            {
                throw new ArgumentException("战斗结果队列配置无效");
            }

            this.battleId = battleId;
            this.capacity = capacity;
            queue = new Queue<BattleTickResult>(capacity);
        }

        /// <summary>在修改战斗状态前检查宿主结果槽位。</summary>
        public void Reserve(long expectedBattleId)
        {
            if (battleId != expectedBattleId || queue.Count >= capacity)
            {
                throw new InvalidOperationException("结果队列归属不符或容量不足");
            }
        }

        /// <summary>保存正常结束的候选步，等待时钟提交水位。</summary>
        internal void Stage(in BattleTickResult result)
        {
            queue.Enqueue(result);
        }

        /// <summary>读取下一个已经被时钟确认的成功步。</summary>
        public bool TryTakeCommitted(long committedTick, out BattleTickResult result)
        {
            if (queue.Count == 0 || queue.Peek().Tick > committedTick)
            {
                result = default;
                return false;
            }

            result = queue.Dequeue();
            return true;
        }
    }

    public sealed class BattleWorld : IDisposable
    {
        private readonly long id;
        private readonly IBattleStages stages;
        private readonly QuadTargetBatch towerIndex, unitIndex;
        private readonly BattleOutbox outbox;
        private readonly int batchSize;
        private long lastCandidate;
        private bool faulted, stopped, disposed;

        public bool Faulted { get { return faulted; } }

        /// <summary>接管本回合两棵索敌树，接入战斗行动所有者与宿主输出。</summary>
        public BattleWorld(long battleId, IBattleStages stages, QuadTargetBatch towerIndex,
            QuadTargetBatch unitIndex, BattleOutbox outbox, int batchSize)
        {
            if (battleId <= 0 || batchSize <= 0 || ReferenceEquals(towerIndex, unitIndex))
            {
                throw new ArgumentException("战斗实例配置无效");
            }

            id = battleId;
            this.stages = stages ?? throw new ArgumentNullException(nameof(stages));
            this.towerIndex = towerIndex ?? throw new ArgumentNullException(nameof(towerIndex));
            this.unitIndex = unitIndex ?? throw new ArgumentNullException(nameof(unitIndex));
            this.outbox = outbox ?? throw new ArgumentNullException(nameof(outbox));
            this.batchSize = batchSize;
        }

        /// <summary>由固定时钟执行一次候选步。发送结果由宿主的提交水位控制。</summary>
        public TickDecision RunCandidate(in BattleTick tick)
        {
            if (disposed || stopped || faulted || tick.BattleId != id
                || tick.Number != lastCandidate + 1 || tick.TickRate <= 0)
            {
                throw new InvalidOperationException("候选 Tick 与战斗生命周期不符");
            }

            try
            {
                outbox.Reserve(id);
                stages.RevokeExpired(tick);
                stages.CommitCommands(tick);
                stages.FreezePreMove(tick);
                SelectTargets(tick, TargetPhase.PreMove, towerIndex);

                stages.StepRvo2AndCommitPositions(tick);
                stages.FreezePostMove(tick);
                SelectTargets(tick, TargetPhase.PostMove, unitIndex);

                stages.CollectAttacks(tick);
                stages.ResolveAndCommit(tick);
                BattleTickResult result = stages.Finish(tick);
                if (result.BattleId != id || result.Tick != tick.Number)
                {
                    throw new InvalidOperationException("候选步输出身份不匹配");
                }

                outbox.Stage(result);
                lastCandidate = tick.Number;
                stopped = result.Decision == TickDecision.StopAfterCommit;
                return result.Decision;
            }
            catch
            {
                // 禁止故障世界继续推进，帧通道保留此前提交水位。
                faulted = true;
                throw;
            }
        }

        /// <summary>完成索敌批并全批预检，按寻敌者稳定顺序交给领域所有者。</summary>
        private void SelectTargets(in BattleTick tick, TargetPhase phase, QuadTargetBatch batch)
        {
            TargetPhaseInput input = stages.CaptureTargeting(tick, phase);
            BatchStamp stamp = input.Stamp;
            if (stamp.BattleId != tick.BattleId || stamp.Tick != tick.Number || stamp.Phase != phase)
            {
                throw new InvalidOperationException("索敌输入与消费阶段不符");
            }

            if (input.Queries.IsEmpty)
            {
                return;
            }

            batch.Begin(stamp, input.Targets.Length, input.Queries.Length, input.Excluded.Length);
            for (int i = 0; i < input.Targets.Length; ++i)
            {
                batch.SetTarget(i, input.Targets[i]);
            }

            for (int i = 0; i < input.Excluded.Length; ++i)
            {
                batch.SetExcluded(i, input.Excluded[i]);
            }

            long previousId = 0;
            for (int i = 0; i < input.Queries.Length; ++i)
            {
                TargetQuery query = input.Queries[i];
                if (query.Seeker.Id <= previousId)
                {
                    throw new InvalidOperationException("寻敌者必须唯一且按稳定 ID 升序");
                }

                previousId = query.Seeker.Id;
                batch.SetQuery(i, query);
            }

            batch.Run(batchSize, forceRebuild: phase == TargetPhase.PostMove);
            for (int i = 0; i < input.Queries.Length; ++i)
            {
                TargetResult result = batch.ReadResult(stamp, i);
                if (!result.Seeker.Matches(input.Queries[i].Seeker) || !stages.CanApplyTarget(stamp, result))
                {
                    throw new InvalidOperationException("索敌结果未通过身份或资格预检");
                }
            }

            for (int i = 0; i < input.Queries.Length; ++i)
            {
                TargetResult result = batch.ReadResult(stamp, i);
                stages.ApplyTarget(stamp, result);
            }
        }

        /// <summary>停止候选步并释放本回合的 Native 批处理工作区。</summary>
        public void Dispose()
        {
            if (disposed)
            {
                return;
            }

            disposed = true;
            try
            {
                unitIndex.Dispose();
            }
            finally
            {
                try
                {
                    towerIndex.Dispose();
                }
                finally
                {
                    stages.Dispose(); // 完成并释放本场 ORCA，以及其余场内工作区。
                }
            }
        }
    }
}
~~~

- FrameClock 的候选步回调传入当场 BattleId、候选 Tick 与 TickRate，调用 RunCandidate。外层时钟只有收到正常返回才提交该 Tick。宿主在一次 Advance 正常返回或发生故障之后，按该时钟的实际 CommittedTick 调用 TryTakeCommitted。Tick 0 由准备流程单独发布。SnapshotKey 指向宿主有界快照仓，槽位释放随消费者完成。Finish 的实现必须先完成 Cleanup，再把可跨世界存续的值保存在对应槽位。
- 第 18 份运行类的 Run 把两个 Job 接成一个依赖链，在消费者前完成。四叉树几何阶段版本由场地提供。塔与单位各有一棵树、一组缓冲。PostMove 在移动系统提交位置之后采集快照。查找到结果时，预检使用该阶段的冻结读视图；没有目标表示撤销或保持空锁，已经通过保持策略的锁定不参加重查。ApplyTarget 对已预检结果按稳定 ID 写入，阶段内不会执行改变其他寻敌者资格的回调。
- 接入 Entitas 时建议把 RunCandidate 主体的调用顺序写成根 Systems 中的固定顺序。工作版也可用一个 IExecuteSystem 调用此编排，前后挂该回合的 Initialize 与 TearDown；每 Tick 只从一个入口执行阶段。Finish 和 Entitas Cleanup 由同一层安排。单位及建筑按 EntityRef 查当前映射，用 Jenny 生成的 ReplaceX/RemoveX 写组件，组件字段与实际 Header 以生成物为准。Release 时先停止输入、补步和发布入口，再完成 Job、释放 RVO2 与 Buff 工作区、退场 Context。 [Entitas 本项目规范](../../../.agents/skills/entitas-dev/references/systems-and-lifecycle.md) · [世界拆卸](14-GameManager与游戏生命周期实现方案.md)

## RVO2 批量移动调用

`IBattleStages` 拥有一场一个的移动适配器及投影缓冲。下面是阶段方法的主体，`_actorWrites` 连接同一战场的权威位置写入。Stage 05 先通过全扫索敌/直接给定 Goal 使用同一接口，后期启用的四叉树只改变两处索敌消费者。

~~~csharp
public void StepRvo2AndCommitPositions(in BattleTick tick)
{
    // 依当前阻挡版本同步静态/动态轮廓；用稳定 ActorId 顺序填输入槽位。
    _movementProjection.SyncActorsAndObstacles(_rvo2);
    int count = _movementProjection.CaptureInputs(_moveInputs);
    _rvo2.Step(tick.TickRate, _moveInputs.AsSpan(0, count), _moveOutputs.AsSpan(0, count));
    for (int i = 0; i < count; ++i)
    {
        MoveOutput result = _moveOutputs[i];
        _actorWrites.WriteMovement(new EntityRef { Id = result.ActorId, Generation = 0 }, result.Position);
    }
}
~~~

采用执行手册的 ActorId 不复用合同，所有输入/输出只对应本场，Generation 固定为 0。若项目以后复用 ID，则用当前真实代际而非此示例常量。各胶水声明跟随 [19](19-寻路模块实现代码.md)，生产世界只装一套移动与帧结果出口。

## Entitas 状态写回示例

下列组件和生成方法是本稿拟定的接线名。已有正式组件时沿用等价字段与生成 API。BattleEntity 要在 Jenny 迁到 Battle Context 后由生成器提供。每场在创建/销毁实体时维护自己的 ID 映射，数据库或表现层不持有该映射。生成方法的签名以 Jenny 实际产出为准。

~~~csharp
using System;
using System.Collections.Generic;
using Entitas;
using Entitas.CodeGeneration.Attributes;
using GameLogic.FixedPoint;
using GameLogic.ProjectB;
using ProjectB.BattleSpatial;

namespace ProjectB.BattleIntegration
{
    [Context("Battle")]
    public sealed class SpatialIdentityComponent : IComponent
    {
        public EntityRef Value;
    }

    [Context("Battle")]
    public sealed class LockedCombatTargetComponent : IComponent
    {
        public EntityRef Value;
    }

    public sealed class BattleEntityWrites
    {
        private readonly IReadOnlyDictionary<long, BattleEntity> entities;

        public BattleEntityWrites(IReadOnlyDictionary<long, BattleEntity> entities)
        {
            this.entities = entities;
        }

        /// <summary>确认业务身份仍对应本回合的现行实体和代际。</summary>
        public bool TryResolve(EntityRef identity, out BattleEntity entity)
        {
            if (identity.Id > 0 && entities.TryGetValue(identity.Id, out entity)
                && entity.isEnabled && entity.hasSpatialIdentity
                && entity.spatialIdentity.Value.Matches(identity))
            {
                return true;
            }

            entity = null;
            return false;
        }

        /// <summary>在阶段预检通过后，通过生成 API 提交单位或建筑锁定。</summary>
        public void WriteTarget(in TargetResult result)
        {
            if (!TryResolve(result.Seeker, out BattleEntity seeker))
            {
                throw new InvalidOperationException("寻敌者已失效");
            }

            if (result.Target.Id != 0)
            {
                seeker.ReplaceLockedCombatTarget(result.Target);
            }
            else if (seeker.hasLockedCombatTarget)
            {
                seeker.RemoveLockedCombatTarget();
            }
        }

        /// <summary>供单位所有者在统一移动提交位调用。</summary>
        public void WriteMovement(EntityRef identity, Point2 position)
        {
            if (!TryResolve(identity, out BattleEntity entity))
            {
                throw new InvalidOperationException("移动提交对象已失效");
            }

            ActorState actor = entity.actor.Value;
            actor.Position = position;
            entity.ReplaceActor(actor);
        }
    }
}
~~~

CanApplyTarget 先用 TryResolve 对照两端身份，再按当前阶段冻结投影校验有效性。空目标只查寻敌者。WriteTarget 和 WriteMovement 由原所有权系统在预检后调用，避免生成组件通知成为一次新的索敌入口。

## 所有者适配时的准确调用

| 所有者入口 | 操作与输入输出 |
| --- | --- |
| StepRvo2AndCommitPositions | 按本场目标、射程和地图点生成 Goal/Stop。登记存活 Agent 和当前障碍，统一调用 `ORCABundle<Agent>.orca.Run(1f / TickRate)`；完成后量化回填坐标。 |
| CollectAttacks | 塔对射程外旧锁按 12 号条件重选。单位和塔均用 PostMove 的位置，以当步冻结的屏障规则计算当前位置发射授权。通过检查的值意图由编排分配 AttackInstanceId 并封成请求。只有获准发射才推进动作和冷却。 |
| ResolveAndCommit | 按命中合同解析同批请求，单位与建筑写其生命和待退出状态。生成下一步 NavRevision 与保护资格差量。N 步击破外塔在 N+1 才启用新规则，当步已采集的合法攻击按共同快照结算。 |

### 接入案例

同一 Tick 内单位 A 选塔 B，目标点是 B 外侧可接敌位置。RVO2 根据当前 Agent 和障碍更新所有单位，再写回权威位置，塔 B 的 PostMove 索敌能看到本步刚进入射程的单位。若 B 在 N 步结算被击破，N+1 的预步资格令旧锁失效，动态障碍更新后用新目标或配置入口推进。候选 N+1 中途报错时，外层只消费此前 CommittedTick 内已封存的快照。

先接 RVO2 与已有帧入口，验证固定阵容后再按实际规模接索敌 Job。重点覆盖移动阶段快照、同数量障碍换位、目标资格变化、暂停/释放和同环境回放。本次为文档修订，编译与 Unity 验收待执行。
