```csharp
// RunRewardService.FinalizeAsync。ProfileRepository 单写入队列承串行提交。
internal async UniTask<RunRewardReceipt> FinalizeAsync(RunOutcome outcome)
{
    PendingReward frozen = _formula.Freeze(outcome);
    await _profiles.WriteAsync(profile => profile.RegisterPending(frozen));
    return await _profiles.TransactAsync(profile =>
    {
        if (profile.TryGetRunReceipt(outcome.RunId, out var saved))
            return ProfileWrite<RunRewardReceipt>.ReadOnly(profile, saved);

        PendingReward pending = profile.GetPendingRun(outcome.RunId);
        RunRewardReceipt receipt = RunRewardReceipt.FromFrozenCreditLines(pending);
        return ProfileWrite<RunRewardReceipt>.Commit(profile.Apply(receipt), receipt);
    });
}
```

`RegisterPending` 写 `(ProfileId, RunId, OutcomeDigest, RewardRulesVersion, FrozenCreditLines)`。已有同摘要 pending/receipt 时返回原档，同 RunId 异摘要报冲突并保留现有记录。Apply 同时记回执、增钱包、移票和累加战绩。启动从持久 pending 恢复，由同一事务消费端处理。暂存与主文件落盘置换按目标平台实现，读取旧/新有效版本时校验版本、Revision 和校验值；反馈“到账”在成功保存后发送。重试重取持久值，不能在内存重复加余额。

```csharp
// NodePurchaseService，三棵树传不同目录与钱包配置。
internal NodeReceipt Buy(ProfileState profile, NodeCommand cmd, out ProfileState next)
{
    next = profile;
    if (profile.TryGetNodeReceipt(cmd.Id, out var existing)) return existing;
    if (!_catalog.TryGet(cmd.Tree, cmd.NodeId, out var node) ||
        !profile.Satisfies(node.Requirements) || !profile.Wallet.CanDebit(node.Price))
        return NodeReceipt.Rejected(cmd.Id);

    var receipt = NodeReceipt.Bought(cmd.Id, node.Key);
    next = profile.WithNode(node, profile.Wallet.Debit(node.Price), receipt);
    return receipt;
}
```

`Satisfies` 包含节点级数上限、实例归属、前置和互斥。事务 Store 再使用预期 Revision 提交。UI 只显示持久成功值。角色解锁复用同一事务通道，校验角色的实际来源与拥有状态，开放白名单才写 Profile。

```csharp
// RunFactory.TryFreeze。冻结目录句柄和 Profile 修订，活跃局持有 lease。
internal bool TryFreeze(in StartRequest req, ProfileState profile,
    ContentLease lease, out RunStartSpec spec)
{
    spec = default;
    if (!profile.OwnsRole(req.RoleId) ||
        !lease.Catalog.TryGetStart(req.RoleId, req.DifficultyId, out var start) ||
        !lease.Catalog.ValidateDisabled(profile.DisabledContent, start))
        return false;

    var upgrades = _loadouts.Build(profile, lease.Catalog, req.RoleId);
    spec = RunStartSpec.Freeze(start, req.Seed, lease, profile.Revision,
        profile.DisabledContent, upgrades);
    return true;
}
```

`RunStartSpec.Freeze` 复制禁用、装备/节点值并增加目录租约引用，建局失败和离局时归还。正式移除某条配置的兼容策略由迁移和开局校验处理。

**阶段验收。** 从干净档先走一次胜利、一次失败、一次取消，回大厅打开结算和全局背包，只有前两者产生正式回执。模拟暂存置换之前、登记后、发奖后断开；登记后换奖励规则版本再启动，钱包仍按冻结条目入账且回执唯一。迁移旧 schema 样本，原有资产仍可见；新角色/科技/禁用只在重开后的快照生效。

```csharp
// ContentCatalogLoader.TryPublishAsync 的核心。
// LoadValidatedAsync 自己负责取消、失败时卸载获得的句柄。
internal async UniTask<bool> TryPublishAsync(int request, CancellationToken token)
{
    ContentLease candidate = await _loader.LoadValidatedAsync(token);
    if (candidate == null) return false;
    if (request != _latestRequest || token.IsCancellationRequested)
    {
        candidate.Dispose();
        return false;
    }

    ContentLease previous = _current;
    _current = candidate;
    previous?.Dispose(); // 退目录的引用；活动 Run 持有的租约仍有效。
    return true;
}
```

```csharp
// ActorViewScope.BindAsync，阶段 04 同一路径。
private async UniTask BindAsync(ActorViewKey key, string address, CancellationToken token)
{
    var instance = await GameModule.Resource.LoadGameObjectAsync(address, _root, token);
    if (instance == null) return;
    if (token.IsCancellationRequested || !_slots.TryGetValue(key, out var slot) ||
        slot.Instance != null)
    {
        UnityEngine.Object.Destroy(instance);
        return;
    }

    slot.Bind(instance);
    slot.Render(slot.LatestState); // 加载期间已提交的位置、阵营等取最新值。
}
```
