## 11 回合间经营

前置：[回合与五层循环](04-卡牌与五层循环.md#s10)。阶段分三批。先让收益进入真实钱包并打开商店，然后接奖励与选择，最后将各来源效果带回战场。每批用既有固定样本走到下一战；经营窗口命令带当前窗口版本和 CommandId。下文交易/领票的方法运行于 `RunSession` 同一个 `_state` 的提交入口，服务文件只构造计划；可将两段提交方法放在 `RunSession` 的 `partial` 文件。计划完成后再赋值和发布。

```csharp
// RoundIncome.Interest。配置保证 step > 0，其他输入非负。
internal long Interest(long openingGold, int step, int perStep, int cap)
{
    return Math.Min(cap, checked(openingGold / step * perStep));
}

// LevelProgress.Grant。tables 保存可达到的最高等级及升级阈值。
internal LevelGain Grant(LevelState before, int xp)
{
    int level = before.Level;
    long remaining = checked(before.Remainder + xp);
    while (_tables.TryGetCostOfNext(level, out int required) && remaining >= required)
    {
        remaining -= required;
        level++;
    }
    return new LevelGain(new LevelState(level, remaining), before.Level + 1, level);
}
```

```csharp
// ShopOfferBuilder 核心。取值用商店随机流的局部副本，提交才换回局状态。
private OfferSpec? Roll(int slot, in ShopSnapshot snapshot, ref RandomStream rng)
{
    _candidates.Clear();
    int total = 0;
    foreach (OfferSpec spec in _catalog.ByStableId)
    {
        if (!spec.IsEligible(slot, snapshot) || spec.Weight <= 0) continue;
        total = checked(total + spec.Weight);
        _candidates.Add(spec);
    }
    if (total == 0) return null;

    int roll = rng.NextInt(total);
    foreach (OfferSpec spec in _candidates)
    {
        if (roll < spec.Weight) return spec;
        roll -= spec.Weight;
    }
    throw new InvalidOperationException("权重表与抽样结果不一致。");
}
```

先应用本窗锁定与硬保底债，再调用普通 Roll。软权重在 `IsEligible` 后求有效权重并保存在同一临时项，循环取权时使用一致值。提交前预检局域可写对象，构造完整钱包/报价/背包/牌组后一次替换，组件与 buff 的细粒度变更计划随状态保存。

```csharp
// TradeService.Execute。TradePlanner 同时准备参与者的下一状态。
internal TradeReceipt Execute(in TradeCommand cmd)
{
    if (_state.TradeReceipts.TryGetValue(cmd.Id, out var saved)) return saved;
    if (!cmd.Window.Matches(_state.Shop.Window)) return TradeReceipt.Stale(cmd.Id);
    if (!_planner.TryBuild(_state, cmd, out TradePlan plan, out TradeReceipt reject)) return reject;

    _state = plan.NextState; // 已含扣/加币、报价消费或拆除、来源变化和回执。
    return plan.Receipt;
}
```

```csharp
// DeckEditService。draft 是当前命令的局部副本，通过才交 RunSession。
internal bool TryCombine(DeckEditDraft draft, long targetId, long consumedId)
{
    if (targetId == consumedId ||
        !draft.TryGetCard(targetId, out var target) ||
        !draft.TryGetCard(consumedId, out var consumed) ||
        target.DefId != consumed.DefId || target.Stars != consumed.Stars ||
        !draft.CanPromote(target))
        return false;

    if (!draft.TryPlanEquipmentOnCombine(target, consumed, out var equipment))
        return false;

    draft.ApplyEquipmentPlan(equipment);
    draft.Promote(targetId);
    draft.Remove(consumedId);
    draft.RecalculateSynergies();
    return true;
}
```

```csharp
// ClaimTicketService，适用于法球和任务；双向海克斯另加选项过滤。
internal ClaimReceipt Claim(in ClaimCommand request)
{
    if (_state.Claims.TryFindReceipt(request.TicketId, out var done)) return done;
    if (!_state.Claims.TryFindOpen(request.TicketId, out var ticket) ||
        !ticket.Accepts(request.Window, request.ChoiceId))
        return ClaimReceipt.Rejected(request.TicketId);

    ClaimPlan plan = _rewards.Prepare(_state, ticket, request.ChoiceId);
    if (!plan.IsValid) return plan.Receipt;
    _state = plan.StateWithReceipt; // 奖励、票据核销、版本在同一份下一状态。
    return plan.Receipt;
}
```

```csharp
// HexRerollService.TryBuild。draft 的 RNG 是可复制的局域值。
internal bool TryReplaceHeld(HexRerollDraft draft, in HexRerollCommand cmd)
{
    RandomStream rng = draft.Rng;
    if (!draft.TryResolveToken(cmd.TokenId, out var token) ||
        !draft.TrySelectHeld(token, cmd.TargetId, ref rng, out var oldHex) ||
        !draft.TryRollReplacement(token, oldHex, ref rng, out var next) ||
        !draft.CanApplyReplacement(oldHex, next))
        return false;

    draft.RevokeReversibleSources(oldHex.SourceId);
    draft.DetachBoundQuest(oldHex.InstanceId);
    draft.RemoveHeld(oldHex.InstanceId);
    draft.GrantNewHeld(next);
    draft.ConsumeToken(token.Id);
    draft.Rng = rng;
    return true;
}
```

```csharp
// StackedAttributeTrack。阈值和效果来自早期样本表或正式内容。
internal void AdvanceTrack(long gain)
{
    _total = checked(_total + gain);
    int level = _calculator.GetLevel(_total);
    if (level == _level) return;
    _level = level;
    _effects.UpsertAttribute(_stableSource, _calculator.ModifierAt(level));
}

// ActiveSkillSystem.Tick。budget 的单位是半个战斗 Tick，正常消耗 2。
internal void Tick(long tick)
{
    bool active = tick < _activeUntilTick;
    _roundBudget = Math.Max(0, _roundBudget - (active ? 1 : 2));
    if (active && tick == _nextRefundTick)
    {
        _fees.RefundClamped(_refundPerSecond);
        _nextRefundTick = checked(_nextRefundTick + _tickRate);
    }
}
```

技能按评审样本从释放 Tick 开始有效，到 `release + 15 × TickRate` 前结束。第一笔回费设 `release + TickRate - 1`，等量间隔，共最多 15 笔；释放 Tick 与加能先后在同一固定时序锁定。技能时钟与回合倒计时由此互不改写，每次注入的费用恢复可供提前结束判定查询。

### 阶段验收

1. 用固定种子走一次准备、对打、评级、收益、法球、双向选择、商店、装配下一战。从前后状态核对每个钱包、物品、敌阵和牌组。
2. 用同一个 CommandId 重放购买、领取、任务事实与回合结算，合法动作的回执可重取，资产变化一次。
3. 验证 15 秒样例的倒计时差 7.5 秒、退款次数与触发 Tick；技能中不攒充能；跨回合/离场后的短期效果从查询里消失。
4. 重复阶段 10 五层用例，确认旧英雄、旧窗口、旧战场的回调不能写当前局。正式配置未给项保留样本输入。
